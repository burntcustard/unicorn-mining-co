import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {gzipSync} from 'node:zlib';
const root='benchmarking/local/cpu-lowering-2026-10-22';
const previous=JSON.parse(readFileSync('benchmarking/experiments/2026-10-22/go-cpu/support/resources-final.json','utf8'));
const measure=(path:string,source=path)=>{
 const data=readFileSync(source);
 return {path,bytes:data.length,gzipBytes:gzipSync(data,{level:1}).length,sha256:createHash('sha256').update(data).digest('hex')};
};
const before=previous.map((r:any)=>r.path==='bin/server'?measure(r.path,`${root}/server-before`):r);
const after=previous.map((r:any)=>measure(r.path));
for(const r of after.filter((r:any)=>r.path!=='bin/server'))assert.deepEqual(r,before.find((b:any)=>b.path===r.path));
writeFileSync(`${root}/resources-before.json`,JSON.stringify(before,null,2)+'\n');
writeFileSync(`${root}/resources-after.json`,JSON.stringify(after,null,2)+'\n');
console.log({unchangedClientResources:after.length-1,beforeServer:before.at(-1),afterServer:after.at(-1)});
