import { readFileSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';

const path = 'src/server/network/session-benchmark_test.go';
const source = readFileSync(path, 'utf8').replace(
  'usage.Utime.Sec*1000000 + usage.Utime.Usec + usage.Stime.Sec*1000000 + usage.Stime.Usec',
  'int64(usage.Utime.Sec)*1000000 + int64(usage.Utime.Usec) + int64(usage.Stime.Sec)*1000000 + int64(usage.Stime.Usec)',
);
const target = resolve('benchmarking/local/cpu-lowering-2026-10-22/portable-session-benchmark_test.go.txt');
writeFileSync(target, source);
writeFileSync('benchmarking/local/cpu-lowering-2026-10-22/386-overlay.json', JSON.stringify({Replace: {[resolve(path)]: target}}));
