import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { cp, mkdir, readFile, readdir, writeFile } from 'node:fs/promises';
import { basename, join } from 'node:path';
import { gzipSync, gunzipSync } from 'node:zlib';

const topic = 'benchmarking/experiments/2026-10-22/input-acknowledgment';
await mkdir(`${topic}/results`, { recursive: true });
await mkdir(`${topic}/sources`, { recursive: true });
const hash = (bytes: Uint8Array) => createHash('sha256').update(bytes).digest('hex');
const median = (values: number[]) => {
  const sorted = values.toSorted((a, b) => a - b);
  const middle = Math.floor(sorted.length / 2);
  return sorted.length % 2 ? sorted[middle] : (sorted[middle - 1] + sorted[middle]) / 2;
};
const p95 = (values: number[]) => values.toSorted((a, b) => a - b)[Math.ceil(values.length * .95) - 1];
const cases: Record<string, unknown[]> = {};
const settings = new Set<string>();
const assets = new Map<string, string>();
for (const [variant, repetitions] of Object.entries({before: 4, current: 4, fixed: 3, recovery: 3, active: 3})) {
  cases[variant] = [];
  for (let repetition = 1; repetition <= repetitions; repetition++) {
    const name = `ack-browser-${variant}-${repetition}`;
    const bytes = await readFile(`benchmarking/local/${name}.json.gz`);
    const run = JSON.parse(gunzipSync(bytes).toString());
    assert(run.completedAt && !run.failure && run.results.length === 1);
    assert.deepEqual(run.counts, [32]);
    assert.deepEqual(run.workloads, ['module']);
    assert(run.seconds === 45 && run.warmup === 15);
    const result = run.results[0];
    assert(result.summaries.length === 32 && result.captures.length === 32);
    assert(result.summaries.every((sample: { errors: number; closes: number }) => sample.errors === 0 && sample.closes === 0));
    assert(result.worldSeeds.every((seed: number) => seed === 25));
    settings.add(JSON.stringify([run.processor, run.logicalCpus, run.browser, run.viewport, run.runnerHash]));
    for (const asset of result.assets) {
      const key = new URL(asset.url).pathname;
      assert(!assets.has(key) || assets.get(key) === asset.sha256);
      assets.set(key, asset.sha256);
    }
    const ack = p95(result.captures.flatMap((capture: { acknowledgments: {ms: number}[] }) => capture.acknowledgments.map(value => value.ms)));
    cases[variant].push({repetition, date: run.date, completedAt: run.completedAt, ackP95Ms: ack, medianFps: median(result.summaries.map((sample: {fps: number}) => sample.fps)), hostCpuBusyPercent: result.hostCpuBusyPercent, rawSha256: hash(bytes)});
    for (const suffix of ['.json.gz', '.log']) await cp(`benchmarking/local/${name}${suffix}`, `${topic}/results/${name}${suffix}`);
  }
}
assert(settings.size === 1, 'All browser captures must use matching host, browser, viewport, runner');
const metrics = Object.fromEntries(Object.entries(cases).map(([variant, runs]) => [variant, {runs, medianRunP95Ms: median(runs.map(run => (run as {ackP95Ms: number}).ackP95Ms))}]));
const tcp = JSON.parse(gunzipSync(await readFile('benchmarking/local/ack-investigation/tcp-active-paired.json.gz')).toString());
const tcpSummary = [4,32].flatMap(players => ['before','after'].map(variant => {
  const runs = tcp.results.filter((run: {players: number; variant: string}) => run.players === players && run.variant === variant);
  assert(runs.length === 3);
  return {players, variant, medianCpuMsPerTick: median(runs.map((run: {cpuMsPerTick: number}) => run.cpuMsPerTick)), medianWorstClientAckP95Ms: median(runs.map((run: {clients: {inputAckP95Ms: number}[]}) => Math.max(...run.clients.map(client => client.inputAckP95Ms))))};
}));
const binaries = {};
for (const name of ['current', 'final']) {
  const bytes = await readFile(`/tmp/unicorn-ack-${name}-server`);
  binaries[name] = {bytes: bytes.length, gzip1Bytes: gzipSync(bytes, {level: 1}).length, sha256: hash(bytes)};
}
const sources = [
  ['/tmp/unicorn-ack-before/src/server/network/game-session.go', 'pre-cpu-game-session.go.txt'],
  ['/tmp/unicorn-game-session-before-receipt-cap.go', 'deployed-game-session.go.txt'],
  ['/tmp/unicorn-game-session-before-recovery.go', 'cap-only-game-session.go.txt'],
  ['/tmp/unicorn-game-session-before-active-recovery.go', 'timer-only-game-session.go.txt'],
  ['src/server/network/game-session.go', 'final-game-session.go.txt'],
  ['src/server/network/snapshot-cadence_test.go', 'final-snapshot-cadence_test.go.txt'],
  ['src/server/network/session-benchmark_test.go', 'final-session-benchmark_test.go.txt'],
  ['benchmarking/tools/live.ts', 'live.ts.txt'],
  ['benchmarking/tools/live-suite.ts', 'live-suite.ts.txt'],
  ['benchmarking/tools/live-browser-probe.ts', 'live-browser-probe.ts.txt'],
  ['benchmarking/tools/browser-gameplay.ts', 'browser-gameplay.ts.txt'],
  ['benchmarking/tools/live-ack-analysis.ts', 'live-ack-analysis.ts.txt'],
  ['benchmarking/tools/go-transport-check.ts', 'go-transport-check.ts.txt'],
  ['/tmp/unicorn-ack-repeat.ts', 'browser-repeat-driver.ts.txt'],
  ['/tmp/unicorn-ack-retain.ts', 'retain.ts.txt'],
];
for (const [source, name] of sources) await cp(source, `${topic}/sources/${name}`);
for (const name of ['live-ack-evidence.json.gz','local-repeats-analysis.json.gz','recovery-analysis.json.gz','active-analysis.json.gz','tcp-paired.json.gz','tcp-paired.log','tcp-active-paired.json.gz','tcp-active-paired.log','browser-repeats.log','recovery-repeats.log','active-repeats.log','tests-final.log','transport-race-final.log','input-response-final.json.gz','input-response-final.log','regressor-failure.log']) {
  await cp(`benchmarking/local/ack-investigation/${name}`, `${topic}/results/${name}`);
}
const profile = await readFile('src/server/default.pgo');
await writeFile(`${topic}/results/summary.json`, JSON.stringify({browserCaptures: 17, variants: metrics, matchedSettings: JSON.parse([...settings][0]), deployedClientAssets: Object.fromEntries(assets), tcpSummary, binaries, pgo: {path: 'src/server/default.pgo', sha256: hash(profile), bytes: profile.length}, validation: {nodeTestFiles: 69, allGoPackages: true, race: true, keyboardResponse: true, build: true, lint: true, formatGo: true}}, null, 2)+'\n');
const manifest = {};
for (const directory of ['results','sources']) {
  for (const name of await readdir(`${topic}/${directory}`)) {
    const bytes = await readFile(`${topic}/${directory}/${name}`);
    manifest[`${directory}/${name}`] = {bytes: bytes.length, sha256: hash(bytes)};
  }
}
await writeFile(`${topic}/manifest.json`, JSON.stringify(manifest, null, 2)+'\n');
