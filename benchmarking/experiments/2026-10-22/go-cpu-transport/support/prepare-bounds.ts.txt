import {readFileSync,writeFileSync} from 'node:fs';
import {resolve} from 'node:path';
const root='benchmarking/local/cpu-game-2026-10-22';
const file='src/server/gameplay/game-collisions.go';
let source=readFileSync(file,'utf8');
source=source.replace('type BodyRecord struct {','type BodyRecord struct {\n LocalBounds collision.AABB\n BoundsReady bool');
const start=source.indexOf('func (g *GameCollisions) findFree('), end=source.indexOf('func (g *GameCollisions) touched(',start);
let free=source.slice(start,end).replaceAll('*4','*8').replaceAll('* 4','* 8').replace('make([]float64, count*8)','make([]float64, count*16)');
const previous='bounds[i*8], bounds[i*8+1], bounds[i*8+2], bounds[i*8+3] = x-reach, x, start.Y+dy/2, reach';
if(!free.includes(previous))throw Error('bounds patch marker missing');
free=free.replace(previous,`${previous}
  left,right,bottom,top:=x-reach,x+reach,start.Y+dy/2-reach,start.Y+dy/2+reach
  if r.BoundsReady && math.Abs(start.X)<1e9 && math.Abs(start.Y)<1e9 {
   box:=r.LocalBounds
   cx,cy:=(box.LowerBound.X+box.UpperBound.X)/2,(box.LowerBound.Y+box.UpperBound.Y)/2
   hx,hy:=(box.UpperBound.X-box.LowerBound.X)/2,(box.UpperBound.Y-box.LowerBound.Y)/2
   sin,cos:=math.Sincos(r.SweepStart.Rotation)
   mx,my:=x+cx*cos-cy*sin,start.Y+dy/2+cx*sin+cy*cos
   padding:=g.catalog.Simulation.Physics.WakeMargin+r.Radius*min(2,math.Abs(r.Object.Rotation-r.SweepStart.Rotation))+1e-7+1e-12*(math.Abs(mx)+math.Abs(my))
   ex,ey:=math.Abs(cos)*hx+math.Abs(sin)*hy+math.Abs(dx)/2+padding,math.Abs(sin)*hx+math.Abs(cos)*hy+math.Abs(dy)/2+padding
   left,right,bottom,top=max(left,mx-ex),min(right,mx+ex),max(bottom,my-ey),min(top,my+ey)
  }
  bounds[i*8],bounds[i*8+4],bounds[i*8+5],bounds[i*8+6]=left,right,bottom,top`);
free=free.replace('right := bounds[a+1] + bounds[a+3]','right := bounds[a+4]');
free=free.replace('dx, dy := bounds[a+1]-bounds[b+1], bounds[a+2]-bounds[b+2]',`if bounds[a+5]>bounds[b+6] || bounds[b+5]>bounds[a+6] {continue}
   dx, dy := bounds[a+1]-bounds[b+1], bounds[a+2]-bounds[b+2]`);
source=source.slice(0,start)+free+source.slice(end);
source=source.replace('\trecord.Radius = 0\n',` record.Radius = 0
 record.BoundsReady=len(fixtures)>0
 record.LocalBounds=collision.AABB{LowerBound:Vec.Create(math.Inf(1),math.Inf(1)),UpperBound:Vec.Create(math.Inf(-1),math.Inf(-1))}
`);
source=source.replace('record.Radius = max(record.Radius, f.Shape.Base().Radius+extent)',`record.Radius = max(record.Radius, f.Shape.Base().Radius+extent)
  var local collision.AABB
  f.Geometry.ComputeAABB(&local,Vec.Transform(0,0,0))
  record.LocalBounds.Combine(record.LocalBounds,local)`);
const destination=resolve(`${root}/bounds-game-collisions.go.txt`);
writeFileSync(destination,source);
writeFileSync(`${root}/bounds-overlay.json`,JSON.stringify({Replace:{[resolve(file)]:destination}},null,2));
