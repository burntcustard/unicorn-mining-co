import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
import { gunzipSync } from 'node:zlib';

const path = process.argv[2];
const raw = readFileSync(path);
const capture = JSON.parse(
  (path.endsWith('.gz') ? gunzipSync(raw) : raw).toString(),
);
const median = (values: number[]) =>
  values.toSorted((a, b) => a - b)[Math.floor(values.length / 2)];
const groups = Map.groupBy(
  capture.results,
  (r: any) => `${r.players}/${r.workload}`,
);
const fields = [
  'cpuMsPerTick',
  'p95Ms',
  'p99Ms',
  'maxMs',
  'rssBytes',
  'allocationsPerTick',
  'allocatedBytesPerTick',
];
const cases = [];

for (const [name, samples] of groups) {
  const variants = Object.fromEntries(
    ['before', 'after'].map((variant) => {
      const rows = samples.filter((s: any) => s.variant === variant);

      return [
        variant,
        Object.fromEntries(
          fields.map((field) => [
            field,
            median(rows.map((r: any) => r[field])),
          ]),
        ),
      ];
    }),
  );

  for (let i = 0; i < capture.repetitions; i++) {
    const a: any = samples.find(
      (r: any) => r.variant === 'before' && r.repetition === i,
    );
    const b: any = samples.find(
      (r: any) => r.variant === 'after' && r.repetition === i,
    );

    for (const field of [
      'contacts',
      'events',
      'entities',
      'packets',
      ...(capture.outcomeComparison === 'strict' ? ['bytes'] : []),
      'states',
      'nearZeroCollisions',
    ]) {
      assert.deepEqual(a[field], b[field], `${name} ${i} ${field}`);
    }
  }

  const before = variants.before.cpuMsPerTick;
  const after = variants.after.cpuMsPerTick;

  cases.push({
    name,
    before: variants.before,
    after: variants.after,
    cpuSavedPercent: 100 * (1 - after / before),
    speedup: before / after,
  });
}

const before = cases.reduce((sum, c) => sum + c.before.cpuMsPerTick, 0);
const after = cases.reduce((sum, c) => sum + c.after.cpuMsPerTick, 0);
let state = 225;
const random = () =>
  (state = (Math.imul(state, 1664525) + 1013904223) >>> 0) / 2 ** 32;
const bootstraps = [];

for (let iteration = 0; iteration < 10000; iteration++) {
  let a = 0,
    b = 0;

  for (const samples of groups.values()) {
    const pairs = Array.from({ length: capture.repetitions }, () =>
      Math.floor(random() * capture.repetitions),
    );

    a += median(
      pairs.map(
        (i) =>
          samples.find((r: any) => r.variant === 'before' && r.repetition === i)
            .cpuMsPerTick,
      ),
    );
    b += median(
      pairs.map(
        (i) =>
          samples.find((r: any) => r.variant === 'after' && r.repetition === i)
            .cpuMsPerTick,
      ),
    );
  }

  bootstraps.push(100 * (1 - b / a));
}

bootstraps.sort((a, b) => a - b);

const summary = {
  source: path,
  ticks: capture.ticks,
  repetitions: capture.repetitions,
  cases,
  aggregateCpuSavedPercent: 100 * (1 - after / before),
  aggregateSpeedup: before / after,
  equalCaseMeanCpuSavedPercent:
    cases.reduce((s, c) => s + c.cpuSavedPercent, 0) / cases.length,
  equalCaseMeanSpeedup: cases.reduce((s, c) => s + c.speedup, 0) / cases.length,
  bootstrap95Percent: [bootstraps[250], bootstraps[9750]],
  exactComparedOutcomes: true,
};

writeFileSync(
  path.replace(/\.json(?:\.gz)?$/, '-summary.json'),
  JSON.stringify(summary, null, 2),
);

for (const c of cases) {
  console.log(
    `| ${c.name.replace('/', ' | ')} | ${c.before.cpuMsPerTick.toFixed(6)} | ${c.after.cpuMsPerTick.toFixed(6)} | ${c.cpuSavedPercent.toFixed(2)}% |`,
  );
}

console.log(
  JSON.stringify({
    aggregate: summary.aggregateCpuSavedPercent,
    speedup: summary.aggregateSpeedup,
    mean: summary.equalCaseMeanCpuSavedPercent,
    meanSpeedup: summary.equalCaseMeanSpeedup,
    interval: summary.bootstrap95Percent,
  }),
);
