import { readFileSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
const root = 'benchmarking/local/cpu-game-2026-10-22';
const file = 'src/server/physics/body.go';
let source = readFileSync(file, 'utf8');
source = source.replace('Vec.SetTransform(&b.Xf, position, angle)\n\tb.poseVersion++', 'previous := b.Xf\n\tVec.SetTransform(&b.Xf, position, angle)\n\tif b.Xf != previous { b.poseVersion++ }');
source = source.replace('b.Sweep.GetTransform(&b.Xf, 1)\n\tb.poseVersion++', 'previous := b.Xf\n\tb.Sweep.GetTransform(&b.Xf, 1)\n\tif b.Xf != previous { b.poseVersion++ }');
const destination = resolve(`${root}/pose-body.go.txt`);
writeFileSync(destination, source);
writeFileSync(`${root}/pose-overlay.json`, JSON.stringify({Replace: {[resolve(file)]: destination}}, null, 2));
