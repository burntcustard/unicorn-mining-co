import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
import { gunzipSync } from 'node:zlib';
const root='benchmarking/local/cpu-game-2026-10-22';
const file=`${root}/acceptance-transport-final.json.gz`;
const capture=JSON.parse(gunzipSync(readFileSync(file)).toString());
assert(capture.completedAt);
assert.equal(capture.results.length,80);
const median=(values:number[])=>values.toSorted((a,b)=>a-b)[Math.floor(values.length/2)];
const groups=Map.groupBy(capture.results,(r:any)=>`${r.players}/${r.activity}`);
const cases=[];
for(const[name,rows]of groups){
 const variants=Object.fromEntries(['before','after'].map(variant=>{
  const samples=rows.filter((r:any)=>r.variant===variant);
  assert.equal(samples.length,5);
  for(const r of samples){assert.equal(r.gomaxprocs,2);assert.equal(r.gogc,'200');assert.equal(r.affinity,'0,1');assert.equal(r.clients.length,r.players);}
  return[variant,{
   cpuMsPerTick:median(samples.map((r:any)=>r.cpuMsPerTick)),
   userMsPerTick:median(samples.map((r:any)=>r.userMs/r.ticks)),systemMsPerTick:median(samples.map((r:any)=>r.systemMs/r.ticks)),
   allocationsPerTick:median(samples.map((r:any)=>r.allocationsPerTick)),allocatedBytesPerTick:median(samples.map((r:any)=>r.allocatedBytesPerTick)),
   medianWorstInputAckP95Ms:median(samples.map((r:any)=>Math.max(...r.clients.map(c=>c.inputAckP95Ms)))),
   medianWorstSnapshotGapP95Ms:median(samples.map((r:any)=>Math.max(...r.clients.map(c=>c.snapshotGapP95Ms)))),
   minimumSnapshotHz:Math.min(...samples.flatMap((r:any)=>r.clients.map(c=>c.snapshotHz))),
   minimumInputHz:Math.min(...samples.flatMap((r:any)=>r.clients.map(c=>c.inputHz))),
   maximumSnapshotGapMs:Math.max(...samples.flatMap((r:any)=>r.clients.map(c=>c.maxSnapshotGapMs))),
   maximumInputAckMs:Math.max(...samples.flatMap((r:any)=>r.clients.map(c=>c.maxInputAckMs))),
   acknowledgmentsPerClientPerSecond:median(samples.map((r:any)=>r.clients.reduce((s,c)=>s+c.acknowledgments,0)/r.players*1000/r.wallMs)),
  }];
 }));
 cases.push({name,...variants,cpuSavedPercent:100*(1-variants.after.cpuMsPerTick/variants.before.cpuMsPerTick)});
}
let randomState=225;
const random=()=> (randomState=(Math.imul(randomState,1664525)+1013904223)>>>0)/2**32;
const draws=[];
for(let n=0;n<10000;n++){
 let before=0,after=0;
 for(const rows of groups.values()){
  const pairs=Array.from({length:5},()=>Math.floor(random()*5));
  before+=median(pairs.map(i=>rows.find((r:any)=>r.variant==='before'&&r.repetition===i).cpuMsPerTick));
  after+=median(pairs.map(i=>rows.find((r:any)=>r.variant==='after'&&r.repetition===i).cpuMsPerTick));
 }
 draws.push(100*(1-after/before));
}
draws.sort((a,b)=>a-b);
const result={source:file,cases,aggregateCpuSavedPercent:100*(1-cases.reduce((s,c)=>s+c.after.cpuMsPerTick,0)/cases.reduce((s,c)=>s+c.before.cpuMsPerTick,0)),equalCaseMeanCpuSavedPercent:cases.reduce((s,c)=>s+c.cpuSavedPercent,0)/cases.length,bootstrap95Percent:[draws[250],draws[9750]]};
writeFileSync(`${root}/transport-summary.json`,JSON.stringify(result,null,2)+'\n');
for(const c of cases)console.log(`| ${c.name.replace('/',' | ')} | ${c.before.cpuMsPerTick.toFixed(4)} | ${c.after.cpuMsPerTick.toFixed(4)} | ${c.cpuSavedPercent.toFixed(2)}% | ${c.before.medianWorstInputAckP95Ms.toFixed(2)} | ${c.after.medianWorstInputAckP95Ms.toFixed(2)} |`);
console.log(JSON.stringify({aggregate:result.aggregateCpuSavedPercent,mean:result.equalCaseMeanCpuSavedPercent,interval:result.bootstrap95Percent}));
