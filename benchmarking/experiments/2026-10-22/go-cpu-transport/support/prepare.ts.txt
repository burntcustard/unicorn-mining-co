import {readFileSync,writeFileSync,readdirSync,mkdirSync,copyFileSync} from 'node:fs';
import {dirname} from 'node:path';
import {createHash} from 'node:crypto';
const root='benchmarking/local/cpu-game-2026-10-22';
const hash=createHash('sha256');
for(const relative of readdirSync('src/server',{recursive:true,encoding:'utf8'}).filter(p=>p.endsWith('.go')).sort()){
 const path=`src/server/${relative}`, data=readFileSync(path), destination=`${root}/before/${path}.txt`;
 mkdirSync(dirname(destination),{recursive:true});writeFileSync(destination,data);
 if(!relative.endsWith('_test.go'))hash.update(`${path}\0`).update(data).update('\0');
}
copyFileSync('src/server/default.pgo',`${root}/before.pgo`);
copyFileSync('bin/server',`${root}/server-before`);
copyFileSync('/tmp/unicorn-lower-after.test','/tmp/unicorn-game-before.test');
copyFileSync('/tmp/unicorn-lower-transport-after','/tmp/unicorn-game-transport-before');
writeFileSync(`${root}/baseline.json`,JSON.stringify({productionSourceSha256:hash.digest('hex'),profileSha256:createHash('sha256').update(readFileSync(`${root}/before.pgo`)).digest('hex')},null,2)+'\n');
