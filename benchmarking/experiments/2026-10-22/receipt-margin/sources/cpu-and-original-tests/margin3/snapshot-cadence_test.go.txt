func TestSnapshotCadence(t *testing.T) {
	t.Parallel()

	catalog, err := specs.Load()

	if err != nil {
		t.Fatal(err)
	}

	session := NewGameSession(25, catalog)
	socket := &benchmarkSocket{}
	session.Receive(protocol.Control{Type: "hello"}, socket)

	if socket.packets != 2 || socket.sequence != 1 {
		t.Fatal("hello must immediately send welcome and load")
	}

	player := session.playersBySocket[socket]

	acknowledge := func() {
		session.Receive(protocol.Control{Type: "snapshotAck", Sequence: socket.sequence}, socket)
	}

	for tick := uint64(1); tick <= 8; tick++ {
		acknowledge()
		session.Receive(protocol.Control{Type: "input", Tick: session.World.Tick, Sequence: tick, Input: protocol.Input{Thrust: float64(tick % 2)}}, socket)
		session.Tick(1)

		if socket.packets != int(tick)+2 || player.snapshotSequence != tick+1 || player.lastSequence != tick || session.World.Tick != tick {
			t.Fatalf("tick %d: packets=%d sequence=%d input=%d world=%d", tick, socket.packets, socket.sequence, player.lastSequence, session.World.Tick)
		}
	}

	acknowledge()
	session.Tick(3)

	if session.World.Tick != 11 || socket.packets != 11 || player.snapshotSequence != 10 {
		t.Fatal("a catch-up batch must send only its completed state")
	}

	// Receipt batching must retain the original nine-snapshot healthy window.
	window := player.snapshotWindow
	expectedPackets := socket.packets + window - len(player.pendingSnapshots)

	for i := 0; i < 20; i++ {
		session.Tick(1)
	}

	if socket.packets != expectedPackets || len(player.pendingSnapshots) != window {
		t.Fatal("receipts must bound pending snapshots")
	}

	session.Receive(protocol.Control{Type: "snapshotAck", Sequence: player.snapshotSequence + 1}, socket)

	if len(player.pendingSnapshots) != window {
		t.Fatal("an unsent sequence cannot free capacity")
	}

	acknowledge()
	previousSequence := player.snapshotSequence
	session.Tick(1)

	if socket.packets != expectedPackets+1 || player.snapshotSequence != previousSequence+1 || session.World.Tick != 32 {
		t.Fatal("acknowledging must resume snapshots on the next simulation tick")
	}
}

func TestSnapshotRoundTrips(t *testing.T) {
	t.Parallel()

	catalog, err := specs.Load()

	if err != nil {
		t.Fatal(err)
	}

	for _, roundTrip := range []uint64{2, 6, 8, 18} {
		session := NewGameSession(25, catalog)
		socket := &benchmarkSocket{}
		session.Receive(protocol.Control{Type: "hello"}, socket)

		type receipt struct{ at, sequence uint64 }

		receipts := []receipt{{roundTrip, socket.sequence}}
		lateSnapshots := 0

		for tick := 0; tick < 90; tick++ {
			for len(receipts) > 0 && receipts[0].at <= session.World.Tick {
				session.Receive(protocol.Control{Type: "snapshotAck", Sequence: receipts[0].sequence}, socket)
				receipts = receipts[1:]
			}

			previous := socket.packets
			previousReceipt := socket.sequence
			session.Tick(1)

			if socket.sequence != previousReceipt {
				receipts = append(receipts, receipt{session.World.Tick + roundTrip, socket.sequence})
			}

			if socket.packets != previous && session.World.Tick > 60 {
				lateSnapshots++
			}
		}

		player := session.playersBySocket[socket]

		if len(player.pendingSnapshots) > player.snapshotWindow {
			t.Fatal("receipts must bound pending snapshots")
		}

		if roundTrip <= 8 {
			if lateSnapshots != 30 || player.snapshotWindow != 9 {
				t.Fatalf("round trip %d: snapshots=%d", roundTrip, lateSnapshots)
			}
		} else if player.snapshotWindow != 2 {
			t.Fatal("slow receipts must keep two slots")
		}
	}
}

func TestSnapshotReceiptJitter(t *testing.T) {
	t.Parallel()

	catalog, err := specs.Load()

	if err != nil {
		t.Fatal(err)
	}

	for _, delays := range [][]uint64{{2, 4, 3, 2}, {2, 8, 3, 1}, {1, 2, 10, 2}, {18}} {
		for _, recoveryDelay := range []uint64{2, 6, 8} {
			t.Run(fmt.Sprintf("%v/recovery-%d", delays, recoveryDelay), func(t *testing.T) {
				session := NewGameSession(25, catalog)
				socket := &benchmarkSocket{}
				session.Receive(protocol.Control{Type: "hello"}, socket)
				player := session.playersBySocket[socket]

				type receipt struct{ at, sequence uint64 }

				receipts := []receipt{{delays[0], socket.sequence}}
				lastReceipt := delays[0]
				requests, snapshots, recovered := 0, 0, 0

				for tick := uint64(0); tick < 300; tick++ {
					for len(receipts) > 0 && receipts[0].at <= session.World.Tick {
						session.Receive(protocol.Control{Type: "snapshotAck", Sequence: receipts[0].sequence}, socket)
						receipts = receipts[1:]
					}

					previous, requested := socket.packets, socket.sequence
					session.Receive(protocol.Control{Type: "input", Tick: session.World.Tick, Sequence: tick + 1, Input: protocol.Input{Turn: float64(int(tick%3) - 1)}}, socket)
					session.Tick(1)

					if player.lastSequence != tick+1 {
						t.Fatal("receipt backpressure delayed an input")
					}

					if len(player.pendingSnapshots) > 9 {
						t.Fatal("receipt batching expanded the healthy snapshot window")
					}

					if len(delays) == 1 && tick < 180 && len(player.pendingSnapshots) > 2 {
						t.Fatal("slow receipts expanded the two-snapshot window")
					}

					if socket.sequence != requested {
						delay := recoveryDelay

						if tick < 180 {
							delay = delays[requests%len(delays)]
						}

						// Immediate receipts on one WebSocket retain their send order.
						lastReceipt = max(lastReceipt, session.World.Tick+delay)
						receipts = append(receipts, receipt{lastReceipt, socket.sequence})
						requests++
					}

					if socket.packets != previous {
						if tick >= 60 && tick < 180 {
							snapshots++
						}

						if tick >= 270 {
							recovered++
						}
					}
				}

				if slices.Max(delays) <= 8 && snapshots != 120 {
					t.Fatalf("healthy jitter lost snapshots: %d", snapshots)
				}

				if recovered != 30 || player.snapshotWindow != 9 {
					t.Fatal("snapshot delivery did not recover after slow receipts")
				}
			})
		}
	}
}

func TestSnapshotBackpressureAfterClientStall(t *testing.T) {
	t.Parallel()

	catalog, err := specs.Load()

	if err != nil {
		t.Fatal(err)
	}

	session := NewGameSession(25, catalog)
	socket := &benchmarkSocket{}
	session.Receive(protocol.Control{Type: "hello"}, socket)
	player := session.playersBySocket[socket]

	for range 30 {
		socket.acknowledge(session)
		session.Tick(1)
	}

	if player.snapshotWindow != 9 || player.ackInterval <= 1 {
		t.Fatal("fast receipts did not enable cumulative batching")
	}

	// A stalled browser cannot decode or acknowledge snapshots, while inputs
	// already sent by another task can still reach the server.
	for sequence := uint64(1); sequence <= 30; sequence++ {
		session.Receive(protocol.Control{Type: "input", Tick: session.World.Tick, Sequence: sequence, Input: protocol.Input{Turn: 1}}, socket)
		session.Tick(1)

		if player.lastSequence != sequence || len(player.pendingSnapshots) > 9 {
			t.Fatal("a client stall delayed input or expanded its snapshot backlog")
		}
	}

	if len(player.pendingSnapshots) != 9 || socket.sequence != player.snapshotSequence {
		t.Fatal("the final bounded snapshot must request a receipt")
	}

	previous := socket.packets
	recoveryStarted := session.World.Tick
	socket.acknowledge(session)
	recoverySeconds := float64(player.snapshotRecoveryUntil-recoveryStarted) * catalog.Simulation.SimulationStep

	if recoverySeconds < 1 || recoverySeconds >= 1+catalog.Simulation.SimulationStep {
		t.Fatal("a delayed receipt must allow one second of recovery")
	}

	session.Tick(1)

	if socket.packets != previous+1 || len(player.pendingSnapshots) != 1 || socket.inputSequence != 30 {
		t.Fatal("the first receipt after a stall must resume current snapshots")
	}

	socket.acknowledge(session)

	if player.snapshotWindow != 2 || player.ackInterval != 1 || player.snapshotRecoveryUntil <= session.World.Tick {
		t.Fatal("one fast receipt must not reopen the healthy window after a stall")
	}

	previousRecovery := player.snapshotRecoveryUntil
	session.Tick(1)
	requested := socket.sequence
	session.Tick(9)
	session.Receive(protocol.Control{Type: "snapshotAck", Sequence: requested}, socket)
	socket.acknowledge(session)

	if player.snapshotRecoveryUntil <= previousRecovery || player.snapshotWindow != 2 {
		t.Fatal("another delayed receipt must extend recovery")
	}

	previousRecovery = player.snapshotRecoveryUntil
	// A catch-up batch emits a fresh snapshot after a period without receipts.
	// Its low receipt lag must not count the quiet period as recovery progress.
	session.Tick(player.snapshotRecoveryUntil - session.World.Tick + 1)
	socket.acknowledge(session)

	if player.snapshotRecoveryUntil <= previousRecovery || player.snapshotWindow != 2 || player.lastSnapshotReceipt != session.World.Tick {
		t.Fatal("a fast receipt after a quiet period must restart recovery")
	}

	for session.World.Tick < player.snapshotRecoveryUntil {
		previous := socket.packets
		session.Tick(1)
		socket.acknowledge(session)

		if socket.packets != previous+1 {
			t.Fatal("timely receipts must preserve snapshot cadence during recovery")
		}

		if session.World.Tick < player.snapshotRecoveryUntil && player.snapshotWindow != 2 {
			t.Fatal("recovery reopened the healthy window before its deadline")
		}
	}

	if player.snapshotWindow != 9 || player.ackInterval != 1 || player.snapshotRecoveryUntil != 0 {
		t.Fatal("a timely receipt after recovery must restore the healthy window")
	}

	token := player.profile.ID
	session.Disconnect(socket)
	session.Receive(protocol.Control{Type: "hello", PlayerToken: token}, socket)

	if player.ackInterval != 0 || player.snapshotRecoveryUntil != 0 || player.lastSnapshotReceipt != 0 {
		t.Fatal("reconnecting retained receipt lag from the previous connection")
	}
}

func TestSnapshotReceiptAfterWindowShrinks(t *testing.T) {
	t.Parallel()

	catalog, err := specs.Load()

	if err != nil {
		t.Fatal(err)
	}

	session := NewGameSession(25, catalog)
	socket := &benchmarkSocket{}
	session.Receive(protocol.Control{Type: "hello"}, socket)
	socket.acknowledge(session)
	session.Tick(1)
	socket.acknowledge(session)
	player := session.playersBySocket[socket]

	for range 6 {
		session.Tick(1)
	}

	requested := socket.sequence
	session.Tick(9)
	session.Receive(protocol.Control{Type: "snapshotAck", Sequence: requested}, socket)

	if player.snapshotWindow != 2 || len(player.pendingSnapshots) != 2 {
		t.Fatal("expected a slow receipt followed by two unrequested snapshots")
	}

	for _, snapshot := range player.pendingSnapshots {
		if snapshot.requested {
			t.Fatal("expected an unrequested tail after the window shrank")
		}
	}

	previous := socket.packets
	session.Tick(1)

	if socket.packets != previous+1 || socket.sequence != player.snapshotSequence {
		t.Fatal("an unrequested tail must allow a current snapshot and receipt request")
	}
}

func TestInputsAcrossServerStall(t *testing.T) {
	t.Parallel()

	catalog, err := specs.Load()

	if err != nil {
		t.Fatal(err)
	}

	session := NewGameSession(25, catalog)
	socket := &benchmarkSocket{}
	session.Receive(protocol.Control{Type: "hello"}, socket)
	player := session.playersBySocket[socket]
	session.Receive(protocol.Control{Type: "snapshotAck", Sequence: socket.sequence}, socket)
	// These edges arrive while the server is stalled. Catch-up must apply them
	// on their own ticks, even when they are more than one catch-up batch ahead.
	session.Receive(protocol.Control{Type: "input", Tick: 10, Sequence: 1, Input: protocol.Input{Turn: 1}}, socket)
	session.Receive(protocol.Control{Type: "input", Tick: 25, Sequence: 2, Input: protocol.Input{Turn: -1}}, socket)

	for tick := 0; tick < 34; tick++ {
		session.Tick(1)
		session.Receive(protocol.Control{Type: "snapshotAck", Sequence: socket.sequence}, socket)
		expected := float64(0)

		if tick >= 10 {
			expected = 1
		}

		if tick >= 25 {
			expected = -1
		}

		if player.ship.Turn != expected {
			t.Fatalf("tick %d: turn=%v, want %v", tick, player.ship.Turn, expected)
		}
	}

	if player.lastSequence != 2 {
		t.Fatal("catch-up must consume both control edges")
	}
}

func TestInputsWindowWrapAndCatchUp(t *testing.T) {
	t.Parallel()

	catalog, err := specs.Load()

	if err != nil {
		t.Fatal(err)
	}

	catalog.Simulation.MaxPredictionTicks = 4
	session := NewGameSession(25, catalog)
	socket := &benchmarkSocket{}
	session.Receive(protocol.Control{Type: "hello"}, socket)
	player := session.playersBySocket[socket]
	sequence := uint64(0)

	for range 6 {
		first := session.World.Tick

		for offset := uint64(0); offset <= 4; offset++ {
			sequence++
			session.Receive(protocol.Control{Type: "input", Tick: first + offset, Sequence: sequence, Input: protocol.Input{Turn: float64(int(sequence%3) - 1)}}, socket)
			// A duplicate must not replace the accepted control edge.
			session.Receive(protocol.Control{Type: "input", Tick: first + offset, Sequence: sequence, Input: protocol.Input{Turn: 5}}, socket)
		}

		session.Tick(3)

		if player.lastSequence != sequence-2 || player.ship.Turn != float64(int((sequence-2)%3)-1) {
			t.Fatal("catch-up changed input order inside the prediction window")
		}

		session.Tick(2)

		if player.lastSequence != sequence || player.ship.Turn != float64(int(sequence%3)-1) {
			t.Fatal("reusing the prediction window lost a future control edge")
		}

		session.Receive(protocol.Control{Type: "snapshotAck", Sequence: socket.sequence}, socket)
	}

	sequence++
	session.Receive(protocol.Control{Type: "input", Tick: session.World.Tick + 4, Sequence: sequence, Input: protocol.Input{Turn: 1}}, socket)
	token := player.profile.ID
	session.Disconnect(socket)
	session.Receive(protocol.Control{Type: "hello", PlayerToken: token}, socket)
	session.Tick(5)

	if player.lastSequence != 0 || player.ship.Turn != 0 {
		t.Fatal("reconnect retained queued input from the previous connection")
	}
}
