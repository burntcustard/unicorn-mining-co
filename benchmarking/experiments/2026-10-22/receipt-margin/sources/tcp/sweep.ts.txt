import { execFileSync } from 'node:child_process';
import { readFileSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { cpuAffinity, goEnvironment } from '../../tools/go-cpu-environment.ts';
const labels = ['sticky', 'margin2', 'margin3', 'margin4'];
const orders = [labels, labels.toReversed(), ['margin2','margin4','sticky','margin3']];
const output = {cpu:cpuAffinity,ticks:300,warmup:120,repetitions:3,players:32,ackDelayMs:0,versions:Object.fromEntries(labels.map(label=>[label,{executable:`/tmp/unicorn-receipt-tcp-${label}`,sha256:createHash('sha256').update(readFileSync(`/tmp/unicorn-receipt-tcp-${label}`)).digest('hex')}])),results:[]};
for(let repetition=0;repetition<orders.length;repetition++)for(const label of orders[repetition]){
 const text=execFileSync(process.execPath,['benchmarking/local/receipt-margin/tcp-client.ts',`/tmp/unicorn-receipt-tcp-${label}`,'-game','-players','32','-ticks','300','-warmup','120','-hz','30'],{encoding:'utf8',env:{...goEnvironment,BENCH_CPU_AFFINITY:cpuAffinity,ACK_DELAY_MS:'0',ACTIVITY:'flight'},stdio:['ignore','pipe','inherit'],timeout:60000});
 const result=JSON.parse(text.trim().split('\n').at(-1));
 output.results.push({label,repetition,...result});
 writeFileSync('benchmarking/local/receipt-margin/tcp-cpu-sweep.json',JSON.stringify(output,null,2)+'\n');
 const avg=key=>result.clients.reduce((sum,c)=>sum+c[key],0)/result.clients.length;
 console.log(JSON.stringify({label,repetition,cpuMsPerTick:result.cpuMsPerTick,receiptsPerSecond:avg('receiptsPerSecond'),worstAckP95:Math.max(...result.clients.map(c=>c.inputAckP95Ms)),worstAckP99:Math.max(...result.clients.map(c=>c.inputAckP99Ms)),longestSnapshotGap:Math.max(...result.clients.map(c=>c.maxSnapshotGapMs))}));
}
