import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { openSync, closeSync, readFileSync, existsSync, readdirSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
const directory=process.cwd()+'/benchmarking/local/native-performance-2026-10-18';
const power=()=>{
  const read=(path:string)=>{try{return readFileSync(path,'utf8').trim()}catch{return null}};
  return {at:new Date().toISOString(),energy:read('/sys/devices/system/cpu/cpu0/cpufreq/energy_performance_preference'),supplies:Object.fromEntries(readdirSync('/sys/class/power_supply').map(name=>[name,{online:read(`/sys/class/power_supply/${name}/online`),status:read(`/sys/class/power_supply/${name}/status`)}]))};
};
let stopping=false;
process.on('SIGTERM',()=>{stopping=true});process.on('SIGINT',()=>{stopping=true});
for(const item of process.argv.slice(2)){
  if(stopping)break;
  const [browser,mode,repeat,scenarios='large,combat,flight',duration='30']=item.split(':');
  assert(['current','old','profile','nocompress','statements','none','pcores','blurOff','models','modelsBlurOff','radar','radarBlurOff','copy','reset','containWorld','containAll','mainGpu'].includes(mode));
  const blurOff=mode==='blurOff'||mode==='modelsBlurOff'||mode==='radarBlurOff';
  const flags=mode==='old'?{GAMEPLAY_OLD_TEXT:'1'}:mode==='profile'?{GAMEPLAY_PROFILE:'1'}:blurOff?{GAMEPLAY_NO_HUD_BLUR:'1'}:['nocompress','statements','none'].includes(mode)?{GAMEPLAY_MINIFY:mode}:{};
  if(mode==='models'||mode==='modelsBlurOff')flags.GAMEPLAY_SHARED_MODELS='1';
  if(mode==='radar'||mode==='radarBlurOff')flags.GAMEPLAY_RADAR_PATH='1';
  if(mode==='copy'||mode==='reset')flags.GAMEPLAY_CLEAR_MODE=mode;
  if(mode==='containWorld'||mode==='containAll')flags.GAMEPLAY_CONTAINMENT=mode==='containWorld'?'world':'all';
  if(mode==='mainGpu')flags.GAMEPLAY_GPU='1';
  const name=`text-${mode}-${browser}-${repeat}`;
  const output=`${directory}/${name}.json`;
  assert(!existsSync(output),`Preserve ${name}`);
  const environment=Object.fromEntries(Object.entries(process.env).filter(([key])=>!key.startsWith('GAMEPLAY_')));
  Object.assign(environment,{CHROME_BIN:'/snap/bin/chromium',FIREFOX_BIN:'/usr/bin/firefox',TMPDIR:`${directory}/tmp`,GAMEPLAY_NATIVE:'1',GAMEPLAY_RESOLUTION:'1920x1200',GAMEPLAY_SCENARIOS:scenarios},flags);
  if(mode==='pcores') environment.FIREFOX_BIN='/tmp/unicorn-firefox-pcores.ts';
  const before=power();assert.equal(before.supplies.AC.online,'1');
  const driver=`${directory}/text-comparison.ts`;
  const source=mode==='old'?`${directory}/v2-shared-firefox-checked-1.source-text.ts`:'src/client/ui/text.ts';
  const hash=createHash('sha256').update(readFileSync(source)).digest('hex');
  writeFileSync(output.replace('.json','.request.json'),JSON.stringify({browser,mode,scenarios,duration,flags,driver,firefoxLauncher:environment.FIREFOX_BIN,textSource:source,textHash:hash,before},null,2));
  console.log(JSON.stringify({starting:name,flags,at:new Date().toISOString()}));
  const log=openSync(output.replace('.json','.log'),'w');
  const child=spawn(process.execPath,[driver,browser,duration,output],{env:environment,stdio:['ignore',log,log]});
  const code=await new Promise(resolve=>child.once('exit',resolve));closeSync(log);
  const after=power();writeFileSync(output.replace('.json','.power.json'),JSON.stringify({before,after},null,2));
  assert.equal(code,0,`Inspect ${name}.log`);
  assert.equal(after.supplies.AC.online,'1');assert.equal(after.energy,before.energy);
  const capture=JSON.parse(readFileSync(output,'utf8'));
  assert.equal(capture.diagnostics.oldText,mode==='old');assert.equal(capture.diagnostics.gpuCanvas,Boolean(flags.GAMEPLAY_GPU));assert.equal(capture.diagnostics.noHudBlur,blurOff);
  assert.equal(Boolean(capture.diagnostics.sharedModels),Boolean(flags.GAMEPLAY_SHARED_MODELS));
  assert.equal(Boolean(capture.diagnostics.radarPath),Boolean(flags.GAMEPLAY_RADAR_PATH));
  assert.equal(capture.diagnostics.clearMode || null,flags.GAMEPLAY_CLEAR_MODE || null);
  assert.equal(capture.diagnostics.containment || null,flags.GAMEPLAY_CONTAINMENT || null);
  assert.equal(capture.diagnostics.minifyMode, flags.GAMEPLAY_MINIFY || 'default');
  for(const result of capture.results.filter(r=>r.scenario!=='reference')){assert.equal(result['hud-canvas'],true);assert.equal(result['hud-blur'],blurOff?'none':'blur(4px)');if(flags.GAMEPLAY_CONTAINMENT)assert.equal(result['containment-applied'],true);}
  console.log(JSON.stringify({finished:name,results:capture.results.map(r=>({scenario:r.scenario,fps:r.fps,minimum:r['fps-windows']?.[0],cpu:r.cpu,p95:r.p95,p99:r.p99}))}));
}
