import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { openSync, closeSync, readFileSync, existsSync, readdirSync, writeFileSync } from 'node:fs';
const directory = process.cwd() + '/benchmarking/local/native-performance-2026-10-18';
const shared = { GAMEPLAY_GPU:'1', GAMEPLAY_SHARED_HUD:'1' };
const deferred = { GAMEPLAY_DEFER_SURFACES:'1', GAMEPLAY_DEFER_SKY:'1' };
const text = { GAMEPLAY_LAZY_TEXT:'1', GAMEPLAY_READY_FONT:'1' };
const modes = {
  control: {},
  shared,
  lazy: {...shared, GAMEPLAY_LAZY_TEXT:'1'},
  ready: {...shared, GAMEPLAY_READY_FONT:'1'},
  deferred: {...shared, ...deferred},
  quality: {...shared, ...deferred, ...text},
  qualityBlurOff: {...shared, ...deferred, ...text, GAMEPLAY_NO_HUD_BLUR:'1'},
  raster: {...shared, GAMEPLAY_RASTER_TEXT:'1', GAMEPLAY_READY_FONT:'1'},
  rasterSoftware: {GAMEPLAY_RASTER_TEXT:'1', GAMEPLAY_READY_FONT:'1'},
  rasterSharedSoftware: {GAMEPLAY_SHARED_HUD:'1', GAMEPLAY_RASTER_TEXT:'1', GAMEPLAY_READY_FONT:'1'},
  offscreen: {...shared, GAMEPLAY_RASTER_TEXT:'1', GAMEPLAY_READY_FONT:'1', GAMEPLAY_OFFSCREEN_TEXT:'1'},
  rasterDeferred: {...shared, ...deferred, GAMEPLAY_RASTER_TEXT:'1', GAMEPLAY_READY_FONT:'1'},
};
const power = () => {
  const read = (path: string) => { try { return readFileSync(path,'utf8').trim(); } catch { return null; } };
  return {at:new Date().toISOString(), energy:read('/sys/devices/system/cpu/cpu0/cpufreq/energy_performance_preference'), supplies:Object.fromEntries(readdirSync('/sys/class/power_supply').map(name=>[name,{online:read(`/sys/class/power_supply/${name}/online`),status:read(`/sys/class/power_supply/${name}/status`),capacity:read(`/sys/class/power_supply/${name}/capacity`)}]))};
};
const fields = {GAMEPLAY_OFFSCREEN_TEXT:'offscreenText',GAMEPLAY_GPU:'gpuCanvas',GAMEPLAY_SHARED_HUD:'sharedHud',GAMEPLAY_LAZY_TEXT:'lazyText',GAMEPLAY_READY_FONT:'readyFont',GAMEPLAY_DEFER_SURFACES:'deferSurfaces',GAMEPLAY_DEFER_SKY:'deferSky',GAMEPLAY_RASTER_TEXT:'rasterText',GAMEPLAY_NO_HUD_BLUR:'noHudBlur'};
let stopping = false;
process.on('SIGTERM', () => { stopping = true; });
process.on('SIGINT', () => { stopping = true; });
for(const item of process.argv.slice(2)) {
  if (stopping) break;
  const [browser,mode,repeat,scenarios='large,combat,flight',duration='30']=item.split(':');
  assert(Object.hasOwn(modes,mode), `Unknown experiment ${mode}`);
  const flags=modes[mode];
  for (const flag of Object.keys(flags)) assert(Object.hasOwn(fields,flag), `Metadata must cover ${flag}`);
  const name=`v2-${mode}-${browser}-${repeat}`;
  const output=`${directory}/${name}.json`;
  assert(!existsSync(output), `Preserve ${name}`);
  const environment=Object.fromEntries(Object.entries(process.env).filter(([key])=>!key.startsWith('GAMEPLAY_')));
  Object.assign(environment,{CHROME_BIN:'/snap/bin/chromium',FIREFOX_BIN:'/usr/bin/firefox',TMPDIR:`${directory}/tmp`,GAMEPLAY_NATIVE:'1',GAMEPLAY_RESOLUTION:'1920x1200',GAMEPLAY_SCENARIOS:scenarios},flags);
  const driver=mode==='control'?'benchmarking/tools/gameplay-performance.ts':`${directory}/diagnostic.ts`;
  const before=power();
  writeFileSync(output.replace('.json','.request.json'),JSON.stringify({browser,mode,repeat,scenarios,duration,driver,flags,before},null,2));
  console.log(JSON.stringify({starting:name,flags,at:new Date().toISOString()}));
  const log=openSync(output.replace('.json','.log'),'w');
  const child=spawn(process.execPath,[driver,browser,duration,output],{env:environment,stdio:['ignore',log,log]});
  const code=await new Promise(resolve=>child.once('exit',resolve));closeSync(log);
  writeFileSync(output.replace('.json','.power.json'),JSON.stringify({before,after:power()},null,2));
  assert.equal(code,0,`Inspect ${name}.log`);
  const capture=JSON.parse(readFileSync(output,'utf8'));
  for(const [flag,field] of Object.entries(fields)) assert.equal(Boolean(capture.diagnostics[field]),Boolean(flags[flag]),`${name}: ${field} must match its request`);
  for(const result of capture.results) {
    if(mode!=='control') {
      assert.equal(result['hud-canvas'],!flags.GAMEPLAY_SHARED_HUD);
      assert.equal(result['hud-blur']==='none',Boolean(flags.GAMEPLAY_NO_HUD_BLUR));
    }
  }
  console.log(JSON.stringify({finished:name,results:capture.results.map(r=>({scenario:r.scenario,fps:r.fps,minimum:r['fps-windows']?.[0],cpu:r.cpu,blur:r['hud-blur'],hud:r['hud-canvas']}))}));
}
