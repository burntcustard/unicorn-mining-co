import { game } from '../game';

type Surface = {
  group: Map<number, Surface>;
  layer: number;
  software: boolean;
  canvas: HTMLCanvasElement;
  ctx: CanvasRenderingContext2D;
  cache?: object;
  scale: number;
  seen: boolean;
  left?: number;
  top?: number;
  css?: string;
};

const surfaces = new WeakMap<object, Map<number, Surface>>();
const active = new Set<Surface>();
const pools = new Map<number, Surface[]>();
const transforms = new Map<HTMLCanvasElement, string>();
let composing = false;
// Native Canvas can move cached artwork through the compositor. WebGL submits
// ordinary geometry directly, but additive light must blend with the sky too.
export const composedSurfaces = (ctx: CanvasRenderingContext2D) =>
  typeof ctx.createPattern === 'function';

export const beginSurfaces = () => {
  composing = true;
  active.forEach((surface) => (surface.seen = false));
};

export const endSurfaces = () => {
  composing = false;
  transforms.forEach((css, canvas) => { canvas.style.transform = css; });
  transforms.clear();

  active.forEach((surface) => {
    if (!surface.seen) {
      surface.canvas.parentNode?.removeChild(surface.canvas);
      active.delete(surface);
      const index = surface.layer * 2 + +surface.software;
      let pool = pools.get(index);

      if (!pool) pools.set(index, (pool = []));

      if (pool.length < 8) {
        pool.push(surface);
      } else {
        surface.group.delete(surface.layer);
      }
    }
  });
};

/*
 * Let the browser move unchanged artwork without rasterizing or resampling it
 * on the foreground canvas. Only managed world frames use these DOM layers;
 * previews keep their ordinary canvas path, including attached tool canvases.
 */
export const renderSurface = ({
  ctx,
  key,
  bounds,
  cache,
  light = false,
  floor = false,
  additive = false,
  draw,
}: {
  ctx: CanvasRenderingContext2D;
  key: object;
  bounds: { left: number; right: number; top: number; bottom: number };
  cache?: object;
  light?: boolean;
  floor?: boolean;
  additive?: boolean;
  draw: (ctx: CanvasRenderingContext2D) => void;
}) => {
  if (
    !composing ||
    (!additive && !composedSurfaces(ctx)) ||
    !game.canvas?.isConnected ||
    ctx.canvas !== game.canvas
  ) {
    return false;
  }

  const matrix = ctx.getTransform();
  const layer = floor ? 0 : light ? 2 : 1;
  const software = Boolean(cache) && !floor;
  let group = surfaces.get(key);

  if (!group) surfaces.set(key, (group = new Map()));
  let surface = group.get(layer);

  if (!cache) {
    // Clip dynamic light/floor surfaces to what can reach the viewport. Static
    // artwork keeps its complete image so camera movement never repaints it.
    const inverse = matrix.inverse();
    const points = [
      [0, 0],
      [game.canvas.width, 0],
      [game.canvas.width, game.canvas.height],
      [0, game.canvas.height],
    ].map(([x, y]) => inverse.transformPoint({ x, y }));

    bounds = {
      left: Math.max(bounds.left, Math.min(...points.map((point) => point.x))),
      right: Math.min(
        bounds.right,
        Math.max(...points.map((point) => point.x)),
      ),
      top: Math.max(bounds.top, Math.min(...points.map((point) => point.y))),
      bottom: Math.min(
        bounds.bottom,
        Math.max(...points.map((point) => point.y)),
      ),
    };

    if (bounds.left >= bounds.right || bounds.top >= bounds.bottom) return true;
  }

  if (!surface) {
    const pool = pools.get(layer * 2 + +software);

    surface = pool?.pop();

    if (surface) {
      surface.group.delete(surface.layer);
      surface.group = group;
      surface.layer = layer;
      surface.cache = undefined;
      surface.left = surface.top = undefined;
      surface.css = undefined;
    } else {
      const canvas = document.createElement('canvas');

      canvas.setAttribute('aria-hidden', 'true');

      surface = {
        group,
        layer,
        software,
        canvas,
        ctx: canvas.getContext('2d', { willReadFrequently: software })!,
        scale: 0,
        seen: false,
      };
    }

    const { canvas } = surface;

    canvas.className =
      'world-surface' +
      (light ? ' world-light' : floor ? ' world-floor' : '') +
      (additive ? ' world-spectrum' : '');
    group.set(layer, surface);
  } else if (!surface.seen) {
    const pool = pools.get(layer * 2 + +surface.software);
    const index = pool?.indexOf(surface) ?? -1;

    if (index >= 0) pool!.splice(index, 1);
  }

  const { canvas, ctx: paint } = surface;
  const scale = game.scale;
  const left = Math.floor(bounds.left * scale - 1) / scale;
  const top = Math.floor(bounds.top * scale - 1) / scale;
  const width = Math.ceil(bounds.right * scale + 1) - left * scale;
  const height = Math.ceil(bounds.bottom * scale + 1) - top * scale;

  if (
    !cache ||
    cache !== surface.cache ||
    scale !== surface.scale ||
    left !== surface.left ||
    top !== surface.top
  ) {
    if (
      width > canvas.width ||
      height > canvas.height ||
      (width < canvas.width / 2 && height < canvas.height / 2) ||
      scale !== surface.scale
    ) {
      canvas.width = Math.ceil(width / 64) * 64;
      canvas.height = Math.ceil(height / 64) * 64;
    }

    paint.resetTransform();
    paint.clearRect(0, 0, canvas.width, canvas.height);
    paint.save();
    paint.translate(-left * scale, -top * scale);
    paint.scale(scale, scale);
    draw(paint);
    paint.restore();
    surface.cache = cache;
    surface.scale = scale;
    surface.left = left;
    surface.top = top;
  }

  if (!canvas.isConnected) game.canvas.before(canvas);
  const ratio = innerWidth / game.canvas.width;

  const css = `matrix(${(matrix.a * ratio) / scale},${(matrix.b * ratio) / scale},${(matrix.c * ratio) / scale},${(matrix.d * ratio) / scale},${(matrix.e + matrix.a * left + matrix.c * top) * ratio},${(matrix.f + matrix.b * left + matrix.d * top) * ratio})`;

  if (surface.css !== css) {
    transforms.set(canvas, css);
    surface.css = css;
  }

  surface.seen = true;
  active.add(surface);
  return true;
};
