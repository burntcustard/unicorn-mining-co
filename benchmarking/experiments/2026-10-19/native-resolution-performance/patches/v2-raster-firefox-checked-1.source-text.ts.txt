import { withAlpha } from '../utilities/color';
import { fontText } from '../utilities/font-text';

let fontReady = false;
const textImages = new WeakMap<
  CanvasRenderingContext2D,
  Map<
    string,
    {
      uses: number;
      image?: HTMLCanvasElement;
      left?: number;
      top?: number;
      text?: string;
    }
  >
>();

export interface RenderTextProps {
  game: any; // Supplies the UI canvas context and scale.
  text: string; // The glyphs to draw.
  x: number; // The text's horizontal anchor.
  y: number; // The text's vertical anchor.
  size?: number; // The glyph scale, defaulting to 0.6.
  align?: number; // Horizontal alignment: -1 left, 0 center, 1 right.
  color?: string; // The glyph color, defaulting to '#fff'.
}

export function renderText({
  game,
  text,
  x,
  y,
  size = 0.6,
  align = -1,
  color = '#fff',
}: RenderTextProps) {
  const { ctx, uiScale } = game;
  const displayText = fontText(text);

  ctx.save();
  ctx.scale(uiScale, uiScale);
  ctx.translate(x, y);
  ctx.scale(size, size);


  const transform = (fontReady ||= document.fonts.check?.('16px Gemetric'))
    ? ctx.getTransform()
    : undefined;
  const scaleX = transform?.a;
  const scaleY = transform?.d;
  let cached;

  if (scaleX > 0 && scaleY > 0 && !transform.b && !transform.c) {
    // Rasterize at the final HUD density and place both native and cached
    // text on whole pixels so the bitmap never needs a second resampling.
    ctx.setTransform(
      scaleX,
      0,
      0,
      scaleY,
      Math.round(transform.e),
      Math.round(transform.f),
    );
    let images = textImages.get(ctx);

    if (!images) textImages.set(ctx, (images = new Map()));
    const readout = /^[\d /-]+$/.test(displayText);
    const key = JSON.stringify([
      readout ? [x, y] : displayText,
      scaleX,
      scaleY,
      align,
      color,
    ]);

    cached = images.get(key);

    if (!cached) {
      // Numeric readouts reuse their canvas as the value changes. Repeated
      // labels amortize the bitmap; other changing text stays native.
      if (images.size === 32) {
        const previous = images.keys().next().value; cached = images.get(previous); images.delete(previous); cached.text = undefined;
      } else cached = { uses: 0 };
    } else {
      images.delete(key);
    }

    images.set(key, cached);

    if (
      cached.text !== displayText
    ) {
      const image = cached.image || document.createElement('canvas');
      const paint = image.getContext('2d', { willReadFrequently: true })!;
      paint.font = '16px Gemetric';
      paint.textAlign = ['left', 'center', 'right'][align + 1];
      const metrics = paint.measureText(displayText);
      const left = Math.floor((-metrics.actualBoundingBoxLeft - 2) * scaleX);
      const top = Math.floor(
        (13 - metrics.actualBoundingBoxAscent - 2) * scaleY,
      );
      const width =
        Math.ceil((metrics.actualBoundingBoxRight + 2) * scaleX) - left;
      const height =
        Math.ceil((13 + metrics.actualBoundingBoxDescent + 2) * scaleY) - top;

      if (image.width !== width) image.width = width;

      if (image.height !== height) image.height = height;
      paint.resetTransform();
      paint.clearRect(0, 0, image.width, image.height);
      paint.translate(-left, -top);
      paint.scale(scaleX, scaleY);
      paint.font = '16px Gemetric';
      paint.textAlign = ['left', 'center', 'right'][align + 1];
      paint.textBaseline = 'alphabetic';
      paint.lineJoin = 'round';
      paint.lineWidth = 2;
      paint.strokeStyle = withAlpha({ color: '#000', alpha: 0.5 });
      paint.fillStyle = color;
      paint.strokeText(displayText, 0, 13);
      paint.fillText(displayText, 0, 13);

      Object.assign(cached, {
        image,
        text: displayText,
        left,
        top,
      });
    }
  }

  if (cached?.image) {
    ctx.resetTransform();
    ctx.drawImage(
      cached.image,
      Math.round(transform.e) + cached.left!,
      Math.round(transform.f) + cached.top!,
    );
  } else {
  ctx.font = '16px Gemetric';
  ctx.textAlign = ['left', 'center', 'right'][align + 1];
  ctx.textBaseline = 'alphabetic';
  ctx.lineJoin = 'round';
  ctx.lineWidth = 2;
  ctx.strokeStyle = withAlpha({ color: '#000', alpha: 0.5 });
  ctx.fillStyle = color;
    ctx.strokeText(displayText, 0, 13);
    ctx.fillText(displayText, 0, 13);
  }

  ctx.restore();
}
