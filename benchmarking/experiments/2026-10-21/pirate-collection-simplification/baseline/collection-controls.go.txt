package npc

import (
	"math"
	"slices"

	"github.com/burntcustard/unicorn-mining-co/src/server/collision/query"
	"github.com/burntcustard/unicorn-mining-co/src/server/collision/shape"
	"github.com/burntcustard/unicorn-mining-co/src/server/objects"
	"github.com/burntcustard/unicorn-mining-co/src/server/simulation"
	Vec "github.com/burntcustard/unicorn-mining-co/src/server/vector"
)

type collectionHull struct {
	position Vec.Vector
	rotation float64
	shape    shape.Shape
	radius   float64
}

type collectionPose struct {
	position  Vec.Vector
	transform Vec.TransformValue
}

const collectionPredictionSteps = 40

/*
collectControls chooses an ordinary engine input that carries the item through
the pickup throat. Passive arrival aims for an eventual stop, which can strand
cargo at the doors. Forecasting the actual hatch and physical hull also avoids
approaches that look close at the ship's center but knock the item away.
*/
func (p *Pilot) collectControls(ship *objects.Ship, destination Movement, obstacles []obstacle) (float64, float64) {
	var center Vec.Vector
	found := false

	for _, segment := range ship.Segments {
		if segment.Catches && segment.Module != nil && !(*segment.TargetHealth() < 1) {
			center = segment.LocalPosition

			if outline := segment.Outline(); outline != nil && len(outline.Points) > 0 {
				offset := Vec.Vector{}

				for _, point := range outline.Points {
					offset = Vec.Add(offset, Vec.Create(point[0], point[1]))
				}

				center = Vec.Add(center, Vec.Scale(offset, 1/float64(len(outline.Points))))
			}

			found = true
			break
		}
	}

	if !found {
		return 0, 0
	}

	// Aim beyond the throat center so slowing cannot leave cargo at its entrance.
	center.X -= 5
	offset := Vec.Subtract(destination.Position, ship.Position)
	bearing := math.Atan2(offset.Y, offset.X)
	forecast := p.forecast.reset(ship, navigationRadius(ship), nil)
	// Collection needs to see a complete turn and pickup. Obstacle avoidance
	// resets this shared storage to its usual, shorter prediction horizon.
	forecast.steps = collectionPredictionSteps
	forecast.path = slices.Grow(forecast.path[:0], forecast.steps+1)[:forecast.steps+1]
	forecast.directions = slices.Grow(forecast.directions[:0], forecast.steps)[:forecast.steps]
	slop := ship.Catalog.Simulation.LinearSlop
	var hullStorage [16]collectionHull
	hull := hullStorage[:0]

	for _, collider := range ship.Hitbox() {
		if collider.Physics != nil && !*collider.Physics || collider.Collides != nil && !*collider.Collides {
			continue
		}

		hull = append(hull, collectionHull{
			position: simulation.RotatePoint(Vec.Subtract(collider.GetPosition(), ship.Position), -ship.Rotation),
			rotation: collider.GetRotation() - ship.Rotation,
			shape:    query.ShapeOf(query.ShapeData{Outline: collider.ShapeOutline, Radius: collider.GetRadius(), CollisionMargin: collider.CollisionMargin}, slop),
		})
	}

	// Include the prepared shape's collision skin in each conservative bound.
	for index := range hull {
		hull[index].radius = hull[index].contactRadius(slop)
	}

	entity, exists := ship.World.Entities.Get(destination.IgnoreID)
	var itemShape shape.Shape
	var itemRotation, itemSpin float64
	contactReach := 60.0
	itemRadius := 0.0

	if exists {
		item := entity.Base()
		contactReach = max(contactReach, navigationRadius(ship)+item.Radius)
		itemRotation, itemSpin = item.Rotation, item.Spin
		var outline [][]float64

		if item.ShapeOutline != nil {
			outline = make([][]float64, len(item.ShapeOutline.Points))

			for index, point := range item.ShapeOutline.Points {
				outline[index] = []float64{point[0], point[1]}
			}
		}

		itemShape = query.ShapeOf(query.ShapeData{Outline: outline, Radius: item.Radius, CollisionMargin: new(0.0)}, slop)
		geometry := itemShape.Base()
		radiusSquared := 0.0

		for _, vertex := range geometry.Vertices[:geometry.Count] {
			radiusSquared = max(radiusSquared, Vec.LengthSquared(vertex))
		}

		itemRadius = math.Sqrt(radiusSquared) + geometry.Radius
	}

	var items [collectionPredictionSteps]collectionPose

	for step := range forecast.steps {
		elapsed := float64(step+1) * predictionStep
		position := Vec.AddScaled(destination.Position, destination.Velocity, elapsed)
		items[step] = collectionPose{position, Vec.Transform(position.X, position.Y, itemRotation+itemSpin*elapsed)}
	}

	var offsets [collectionPredictionSteps]Vec.Vector
	// Angular response is shared across thrust choices. Cache these local
	// transforms for one heading, preparing only steps that need contact checks.
	var transformStorage [collectionPredictionSteps * 16]Vec.TransformValue
	transforms := transformStorage[:0]
	transforms = slices.Grow(transforms, forecast.steps*len(hull))[:forecast.steps*len(hull)]
	var prepared [collectionPredictionSteps]bool
	best := math.Inf(1)
	bestHeading, bestThrust, bestTurn := bearing, 0.0, 0.0

	for _, offset := range []float64{0, -.05, .05, -.1, .1, -.2, .2, -.35, .35, -.5, .5, -.75, .75, -1, 1, -1.5, 1.5, -2, 2, math.Pi} {
		heading := bearing + offset
		turn := turnInput(ship, math.Remainder(heading-ship.Rotation, 2*math.Pi))
		forecast.steer(heading, turn)
		clear(prepared[:])

		for step, direction := range forecast.directions {
			offsets[step] = Vec.Create(center.X*direction.X-center.Y*direction.Y, center.X*direction.Y+center.Y*direction.X)
		}

		for _, thrust := range []float64{0, .25, .5, .75, 1} {
			forecast.project(heading, thrust, turn)

			var costs, remaining [collectionPredictionSteps]float64

			// Contact penalties are nonnegative. A geometric lower bound can
			// eliminate a candidate before its detailed collision checks.
			for step := range forecast.steps {
				elapsed := float64(step+1) * predictionStep
				direction := forecast.directions[step]
				hatch := Vec.Add(forecast.path[step+1], offsets[step])
				error := Vec.Subtract(items[step].position, hatch)
				along, across := Vec.Dot(error, direction), Vec.Cross(direction, error)
				x, y := max(0, math.Abs(along)-2), max(0, math.Abs(across)-4)

				if max(x, y)+4*elapsed >= best {
					costs[step] = math.Inf(1)
					continue
				}

				distance := math.Hypot(x, y)
				costs[step] = distance + 4*elapsed
			}

			minimum := math.Inf(1)

			// A suffix minimum bounds every remaining arrival score while the
			// accumulated contact penalty can only increase.
			for step := forecast.steps - 1; step >= 0; step-- {
				minimum = min(minimum, costs[step])
				remaining[step] = minimum
			}

			closest, penalty := math.Inf(1), 0.0

		candidateSteps:
			for step := range forecast.steps {
				if remaining[step]+penalty >= min(best, closest) {
					break
				}

				direction := forecast.directions[step]
				item := items[step].position

				if exists && Vec.DistanceSquared(item, forecast.path[step+1]) < contactReach*contactReach {
					poses := transforms[step*len(hull) : (step+1)*len(hull)]

					if !prepared[step] {
						rotation := math.Atan2(direction.Y, direction.X)
						sine, cosine := math.Sincos(rotation)

						for index, collider := range hull {
							position := Vec.Create(collider.position.X*cosine-collider.position.Y*sine, collider.position.X*sine+collider.position.Y*cosine)
							poses[index] = Vec.Transform(position.X, position.Y, rotation+collider.rotation)
						}

						prepared[step] = true
					}

					for index, collider := range hull {
						a := poses[index]
						a.P = Vec.Add(forecast.path[step+1], a.P)
						reach := collider.radius + itemRadius + slop

						if Vec.DistanceSquared(a.P, item) > reach*reach {
							continue
						}

						contact, touching := query.ContactShapes(collider.shape, itemShape, a, items[step].transform, slop)

						if touching {
							penalty += 2 * max(.1, contact.Depth)

							if remaining[step]+penalty >= min(best, closest) {
								break candidateSteps
							}
						}
					}
				}

				closest = min(closest, costs[step]+penalty)
			}

			if closest < best {
				best, bestHeading, bestThrust, bestTurn = closest, heading, thrust, turn
			}
		}
	}

	return p.avoid(ship, simulation.DirectionOf(bestHeading), bestThrust, bestTurn, obstacles)
}
