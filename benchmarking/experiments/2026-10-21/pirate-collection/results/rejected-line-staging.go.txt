package npc

import (
	"math"
	"github.com/burntcustard/unicorn-mining-co/src/server/objects"
	"github.com/burntcustard/unicorn-mining-co/src/server/simulation"
	Vec "github.com/burntcustard/unicorn-mining-co/src/server/vector"
)

type collectionApproach struct {
	itemID     int64
	checkedAt  uint64
	distance   float64
	reposition Vec.Vector
}

/*
Back away and approach from another angle when cargo stops entering the scoop.
Items still have to turn and pass through the ship's physical gap themselves.
*/
func (c *collectionApproach) movement(ship *objects.Ship, destination Movement) Movement {
	if !destination.Collect {
		*c = collectionApproach{}
		return destination
	}

	distance := Vec.Distance(ship.Position, destination.Position)

	if c.itemID != destination.IgnoreID {
		*c = collectionApproach{itemID: destination.IgnoreID, checkedAt: ship.World.Tick, distance: distance}

		if item, ok := ship.World.Entities.Get(destination.IgnoreID); ok {
			width := math.Inf(1)

			for _, segment := range ship.Segments {
				if segment.Catches && segment.Module != nil && segment.Module.ModuleBase().Spec.CargoGeometry.Paired {
					low, high := math.Inf(1), math.Inf(-1)

					for _, point := range segment.Outline().Points {
						low, high = min(low, point[1]), max(high, point[1])
					}

					width = high - low
					break
				}
			}

			if outline := item.Base().ShapeOutline; outline != nil {
				projected := func(axis Vec.Vector) float64 {
					low, high := math.Inf(1), math.Inf(-1)

					for _, point := range outline.Points {
						value := Vec.Cross(axis, Vec.Create(point[0], point[1]))
						low, high = min(low, value), max(high, value)
					}

					return high - low
				}

				course := simulation.RotatePoint(Vec.Normalize(Vec.Subtract(destination.Position, ship.Position)), -item.Base().Rotation)

				if projected(course) > width {
					best, axis := math.Inf(1), Vec.Vector{}

					for index, point := range outline.Points {
						next := outline.Points[(index+1)%len(outline.Points)]
						candidate := Vec.Normalize(Vec.Create(next[0]-point[0], next[1]-point[1]))

						if extent := projected(candidate); extent < best {
							best, axis = extent, candidate
						}
					}

					if best <= width {
						c.reposition = simulation.RotatePoint(Vec.Scale(axis, 90), item.Base().Rotation)

						if Vec.Dot(c.reposition, Vec.Subtract(ship.Position, destination.Position)) < 0 {
							c.reposition = Vec.Scale(c.reposition, -1)
						}
					}
				}
			}
		}
	}

	if c.reposition != (Vec.Vector{}) {
		direction := Vec.Scale(Vec.Normalize(c.reposition), -1)
		offset := Vec.Subtract(destination.Position, ship.Position)
		angle := math.Remainder(math.Atan2(direction.Y, direction.X)-ship.Rotation, 2*math.Pi)

		if Vec.Dot(offset, direction) > 0 && math.Abs(Vec.Cross(offset, direction)) < 4 && math.Abs(angle) < .3 {
			c.reposition = Vec.Vector{}
			c.checkedAt, c.distance = ship.World.Tick, distance
		}
	}

	if c.reposition == (Vec.Vector{}) {
		if distance < c.distance-2 || distance > 2*navigationRadius(ship) {
			c.checkedAt, c.distance = ship.World.Tick, distance
		}

		if float64(ship.World.Tick-c.checkedAt)*ship.Catalog.Simulation.SimulationStep >= 2 {
			c.reposition = simulation.RotatePoint(Vec.Create(-100, 80), ship.Rotation)

			if item, ok := ship.World.Entities.Get(destination.IgnoreID); ok {
				// A lengthwise approach lets broadside ore fit through the channel.
				c.reposition = simulation.RotatePoint(Vec.Create(120, 0), item.Base().Rotation)

				if Vec.Dot(c.reposition, Vec.Subtract(ship.Position, destination.Position)) < 0 {
					c.reposition = Vec.Scale(c.reposition, -1)
				}
			}
		}
	}

	if c.reposition != (Vec.Vector{}) {
		direction := Vec.Normalize(c.reposition)
		along := Vec.Dot(Vec.Subtract(ship.Position, destination.Position), direction)
		destination.Position = Vec.AddScaled(destination.Position, direction, max(60, along-80))
		destination.Collect = false
		destination.FollowDistance = -40 / ship.Drag
	}

	return destination
}
