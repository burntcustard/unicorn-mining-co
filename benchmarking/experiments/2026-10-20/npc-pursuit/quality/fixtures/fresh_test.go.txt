package npc

import (
	"fmt"
	"github.com/burntcustard/unicorn-mining-co/src/server/objects"
	"github.com/burntcustard/unicorn-mining-co/src/server/simulation"
	Vec "github.com/burntcustard/unicorn-mining-co/src/server/vector"
	"math"
	"testing"
)

var freshRiskScenarios = []struct {
	seed     uint32
	resource int
	speed    float64
}{{101, 2, 170}, {103, 4, 180}, {107, 1, 190}, {109, 4, 170}, {113, 2, 190}}

func TestFreshRiskReferenceGeometry(t *testing.T) {
	for _, scenario := range freshRiskScenarios {
		t.Run(fmt.Sprintf("seed%d/resource%d/speed%g", scenario.seed, scenario.resource, scenario.speed), func(t *testing.T) {
			ship, rocks, center := generatedCourse(t, courseField{seed: scenario.seed, resource: scenario.resource})

			for _, rock := range rocks {
				rock.Spin = 0
			}

			track := narrowPursuitTrack(t, rocks, center, scenario.speed)
			maximumSpeed, maximumYaw, maximumForce := 0., 0., 0.
			_, previous := track.at(0)

			for now := 1. / 120; now < track.times[len(track.times)-1]; now += 1. / 120 {
				_, velocity := track.at(now)
				maximumSpeed = max(maximumSpeed, Vec.Length(velocity))
				maximumYaw = max(maximumYaw, math.Abs(math.Atan2(Vec.Cross(previous, velocity), Vec.Dot(previous, velocity)))*120)
				force := Vec.Add(Vec.Scale(Vec.Subtract(velocity, previous), 120), Vec.Scale(velocity, ship.Drag))
				maximumForce = max(maximumForce, Vec.Length(force))
				previous = velocity
			}

			if maximumSpeed > ship.MaxSpeed() || maximumYaw > ship.TurnRate*ship.RotationalThrust()/ship.Rules.Flight.SpinDivisor || maximumForce > ship.Rules.Flight.ThrustScale*ship.ForwardThrust()/ship.Mass || !pursuitRouteClear(ship, rocks, track.at, min(300, track.times[len(track.times)-1])) {
				t.Fatal("invalid physical reference")
			}

			t.Logf("rocks=%d points=%d lap=%.2fs speed=%.2f yaw=%.2f force=%.2f", len(rocks), len(track.points), track.times[len(track.times)-1], maximumSpeed, maximumYaw, maximumForce)
		})
	}
}

func TestFreshRiskCurvingPursuit(t *testing.T) {
	for _, physical := range []bool{false, true} {
		for _, scenario := range freshRiskScenarios {
			t.Run(fmt.Sprintf("physical%v/seed%d/resource%d/speed%g", physical, scenario.seed, scenario.resource, scenario.speed), func(t *testing.T) {
				ship, rocks, center := generatedCourse(t, courseField{seed: scenario.seed, resource: scenario.resource})

				for _, rock := range rocks {
					rock.Spin = 0
				}

				track := narrowPursuitTrack(t, rocks, center, scenario.speed)
				var result pursuitResult

				if physical {
					reference := objects.CreatePlayerShip(ship.World, objects.Properties{DefinitionID: "courser", PlayerID: new(int64(2))})
					simulation.AddEntity(ship.World, reference)
					result = runCurvingPursuit(ship, track, reference)

					if reference.Dead {
						t.Fatal("reference ship destroyed")
					}
				} else {
					result = runCurvingPursuit(ship, track)
				}

				t.Logf("mean=%.1f p95=%.1f excess=%.1f escaped=%.2fs asteroids=%d ships=%d stationary=%d dead=%v stranded=%v", result.mean, result.p95, result.excess, result.escaped, result.contacts, result.shipContacts, result.stationary, ship.Dead, ship.Stranded())

				if ship.Dead || ship.Stranded() || result.stationary > 5 || result.contacts > 15 || result.shipContacts > 15 {
					t.Fatal("pilot failed fresh held-out course")
				}
			})
		}
	}
}
