package npc
import (
 "testing"
 "fmt"
 "github.com/burntcustard/unicorn-mining-co/src/server/objects"
 "github.com/burntcustard/unicorn-mining-co/src/server/simulation"
)
func TestFinalHeldOutCurvingPursuit(t *testing.T) {
for _,physical:=range []bool{false,true} {
for _,scenario:=range []struct{seed uint32;resource int;speed float64}{
{2,2,170},{3,1,180},{6,2,190},{10,4,170},{13,2,180},{17,4,190},{31,4,170},
} {
t.Run(fmt.Sprintf("physical%v/seed%d/resource%d/speed%g",physical,scenario.seed,scenario.resource,scenario.speed),func(t *testing.T){
ship,rocks,center:=generatedCourse(t,courseField{seed:scenario.seed,resource:scenario.resource})
for _,rock:=range rocks {rock.Spin=0}
track:=narrowPursuitTrack(t,rocks,center,scenario.speed)
if !pursuitRouteClear(ship,rocks,track.at,300) {t.Fatal("invalid hull reference")}
var result pursuitResult
if physical {
reference:=objects.CreatePlayerShip(ship.World,objects.Properties{DefinitionID:"courser",PlayerID:new(int64(2))})
simulation.AddEntity(ship.World,reference)
result=runCurvingPursuit(ship,track,reference)
if reference.Dead {t.Fatal("reference ship destroyed")}
} else {result=runCurvingPursuit(ship,track)}
t.Logf("mean=%.1f p95=%.1f excess=%.1f escaped=%.2fs asteroids=%d ships=%d stationary=%d dead=%v stranded=%v",result.mean,result.p95,result.excess,result.escaped,result.contacts,result.shipContacts,result.stationary,ship.Dead,ship.Stranded())
if ship.Dead || ship.Stranded() || result.stationary>5 || result.contacts>15 || result.shipContacts>15 {t.Fatal("pilot failed held-out course")}
})
}
}
}
