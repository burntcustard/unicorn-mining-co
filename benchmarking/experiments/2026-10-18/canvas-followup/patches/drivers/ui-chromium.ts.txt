#!/usr/bin/env node
import { spawn } from 'node:child_process';
import { mkdtemp, rm } from 'node:fs/promises';
const directory = await mkdtemp(process.cwd() + '/benchmarking/local/fps-followup-2026-10-18/tmp/ui-chromium-');
const args = process.argv.slice(2).map(argument => argument.startsWith('--user-data-dir=') ? '--user-data-dir=' + directory : argument);
const child = spawn('/snap/bin/chromium', args, { stdio: 'inherit' });
process.on('SIGTERM', () => child.kill('SIGTERM'));
process.on('SIGINT', () => child.kill('SIGINT'));
try {
  const code = await new Promise((resolve, reject) => { child.once('exit', resolve); child.once('error', reject); });
  process.exitCode = typeof code === 'number' ? code : 0;
} finally {
  await rm(directory, { recursive: true, force: true });
}
